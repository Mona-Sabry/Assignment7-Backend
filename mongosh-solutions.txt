{ ok: 1 }

 db.createCollection( "logs", { capped: true, size: 1000000 } )

 //15. Delete all books published before 2000.

 //16. Using aggregation Functions, Filter books published after 2000 and sort themby year descending.
db.books.aggregate([
  {
    $match:{
      year:{$gt:2000}
    }
  },
    {
      $sort:{
        year:-1
      }
    }
])


 //17. Using aggregation functions, Find all books published after the year 2000. For each matching book, show only the title, author, and year fields.
db.books.aggregate([
  {
    $match:{
      year:{$gt:2000}
    }
  },
    {
      $project:{
      title:1,
      author:1,
      year:1
      }
    }
])


 //18. Using aggregation functions,break an array of genres into separate documents.
db.books.aggregate( [
  { $match: { title: "Brave New World" } },
  { $project: { _id: 0, title: 1, genres: 1 } },
  { $unwind: "$genres" }
] )


 //19. Using aggregation functions, Join the books collection with the logs collection
db.books.aggregate([
  {
    $lookup: {
      from: "logs",
      let: { bookId: { $toString: "$_id" } },
      pipeline: [
        {
          $match: {
            $expr: {
              $eq: ["$book_id", "$$bookId"]
            }
          }
        }
      ],
      as: "logs"
    }
    },
  {
    $unwind:"$logs"
  },
  {
    $project:{
       _id:0,
      action:"$logs.action",
      book_datails:[{
        title:1,
        author:1,
        year:1
      }]  
    }
  }
